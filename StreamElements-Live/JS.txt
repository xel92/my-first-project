(() => {
  const origin = 'https://xel92.github.io';
  const frame = document.getElementById('dragon-live');
  let ready = false;
  let configuration = null;
  const pending = [];
  function send(type, detail) {
    frame.contentWindow.postMessage({channel:'dragon-se-v1', type, detail}, origin);
  }
  window.addEventListener('onWidgetLoad', event => {
    // Forward settings only; no channel/session credentials are needed.
    configuration = {fieldData: event.detail?.fieldData || {}};
    if (ready) send('onWidgetLoad', configuration);
  });
  window.addEventListener('onEventReceived', event => {
    const detail = event.detail || {};
    if (detail.listener !== 'follower-latest' && !(detail.listener === 'widget-button' && detail.event?.field === 'testFollower')) return;
    const safe = {listener: detail.listener, event: {name:detail.event?.name, displayName:detail.event?.displayName, field:detail.event?.field}};
    if (ready && configuration) send('onEventReceived', safe);
    else pending.push(safe);
  });
  function flush() {
    if (!ready || !configuration) return;
    send('onWidgetLoad', configuration);
    while (pending.length) send('onEventReceived', pending.shift());
  }
  window.addEventListener('onWidgetLoad', flush);
  window.addEventListener('message', event => {
    if(event.source !== frame.contentWindow || event.origin !== origin || event.data?.channel !== 'dragon-se-v1' || event.data.type !== 'ready') return;
    ready = true;
    flush();
  });
  frame.src = origin + '/my-first-project/?overlay=1&se=1&v=' + Date.now();
})();
